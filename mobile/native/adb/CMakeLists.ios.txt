# Copied into android-tools/vendor by tool/build_ios_adb.py.
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(_DARWIN_C_SOURCE __DARWIN_C_LEVEL=__DARWIN_C_FULL _FILE_OFFSET_BITS=64)
add_compile_options(-Wno-attributes)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# Dependency command-line tools and link probes need ordinary executable targets.
set(CMAKE_MACOSX_BUNDLE OFF)
find_package(Threads REQUIRED)

add_subdirectory(boringssl EXCLUDE_FROM_ALL)
add_subdirectory(fmtlib EXCLUDE_FROM_ALL)

set(LZ4_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(LZ4_BUILD_LEGACY_LZ4C OFF CACHE BOOL "" FORCE)
add_subdirectory("${NKAS_ADB_SOURCE}/external/lz4/contrib/cmake_unofficial" "${CMAKE_BINARY_DIR}/lz4" EXCLUDE_FROM_ALL)
set(ZSTD_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${NKAS_ADB_SOURCE}/external/zstd/build/cmake" "${CMAKE_BINARY_DIR}/zstd" EXCLUDE_FROM_ALL)
set(BROTLI_DISABLE_TESTS ON CACHE BOOL "" FORCE)
add_subdirectory("${NKAS_ADB_SOURCE}/external/brotli" "${CMAKE_BINARY_DIR}/brotli" EXCLUDE_FROM_ALL)

set(protobuf_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(protobuf_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(protobuf_BUILD_PROTOC_BINARIES OFF CACHE BOOL "" FORCE)
set(protobuf_BUILD_LIBPROTOC OFF CACHE BOOL "" FORCE)
set(protobuf_WITH_ZLIB OFF CACHE BOOL "" FORCE)
set(ABSL_PROPAGATE_CXX_STD ON CACHE BOOL "" FORCE)
add_subdirectory("${NKAS_ADB_SOURCE}/external/protobuf" "${CMAKE_BINARY_DIR}/protobuf" EXCLUDE_FROM_ALL)

include_directories("${CMAKE_CURRENT_BINARY_DIR}" "${NKAS_GENERATED_PROTO_DIR}"
    "${NKAS_ADB_SOURCE}/porting/adb/include")
include(CMakeLists.libbase.txt)
include(CMakeLists.adb-ios.txt)
include(CMakeLists.zip-ios.txt)
target_include_directories(libadb PRIVATE core/diagnose_usb/include
    "${NKAS_ADB_SOURCE}/external/lz4/lib"
    "${NKAS_ADB_SOURCE}/external/zstd/lib"
    "${NKAS_ADB_SOURCE}/external/brotli/c/include")
target_link_libraries(libadb_crypto_defaults PUBLIC protobuf::libprotobuf)
target_link_libraries(libadb PUBLIC libadb_crypto_defaults libadb_tls_connection_defaults
    libbase libcrypto libcutils libdiagnoseusb liblog libzip ssl crypto decrepit
    protobuf::libprotobuf brotlicommon-static brotlidec-static brotlienc-static
    lz4_static libzstd_static Threads::Threads z "-framework CoreFoundation" "-framework Security")

# The link command identifies the exact archives to merge and checks unresolved symbols.
add_executable(nkas_adb_link_check "${NKAS_BINDING_DIR}/link_check.cpp")
target_include_directories(nkas_adb_link_check PRIVATE "${NKAS_BINDING_DIR}")
target_link_libraries(nkas_adb_link_check PRIVATE libadb)
set_target_properties(nkas_adb_link_check PROPERTIES MACOSX_BUNDLE FALSE)
